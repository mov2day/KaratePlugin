### pets/pets.feature
Feature: Pets API
  Generated from petstore (openapi)

  Background:
    * url baseUrl
    * call read('classpath:common/auth.feature')

  @positive @op=listPets @domain=pets
  Scenario: List pets - returns 200
    # source: GET /pets
    Given path 'pets'
    And param limit = 10
    When method get
    Then status 200
    And match header Content-Type contains 'application/json'
    And match header X-Total-Count == '##regex [0-9]+'
    And match header ETag == '#string'
    And assert response.length <= 100
    And assert new Set(response.map(x => JSON.stringify(x))).size == response.length
    And assert response.length <= 10
    And match each response == read('classpath:schemas/pets.json')

  @positive @op=createPet @domain=pets
  Scenario: Create a pet - returns 201
    # source: POST /pets
    * def payload = karate.copy(read('classpath:data/pets.json')['createPet'])
    * set payload.name = 'Sample name-' + karate.uuid()
    Given path 'pets'
    And request payload
    When method post
    Then status 201
    And match header Content-Type contains 'application/json'
    And match header Location == '#string'
    And match response.kind == 'pet'
    And match response == read('classpath:schemas/pets.json')

  @negative @op=createPet @domain=pets
  Scenario: Create a pet - rejects missing required field 'name' with 400
    # source: POST /pets
    * def payload = karate.copy(read('classpath:data/pets.json')['createPet'])
    * set payload.name = 'Sample name-' + karate.uuid()
    * remove payload.name
    Given path 'pets'
    And request payload
    When method post
    Then status 400
    And match header Content-Type contains 'application/json'
    And match response == read('classpath:schemas/pets-create-pet-400.json')

  @positive @op=getPet @domain=pets
  Scenario: Get a pet - returns 200
    # source: GET /pets/{petId}
    # Assumes a resource with this id exists in the target environment
    * def petId = 1
    Given path 'pets', petId
    When method get
    Then status 200
    And match header Content-Type contains 'application/json'
    And match header ETag == '##string'
    And match response == read('classpath:schemas/pets.json')

  @negative @op=getPet @domain=pets
  Scenario: Get a pet - returns 404 for an unknown id
    # source: GET /pets/{petId}
    * def petId = 999999999
    Given path 'pets', petId
    When method get
    Then status 404
    And match header Content-Type contains 'application/json'
    And match response == read('classpath:schemas/pets-create-pet-400.json')
